# Meal Cart + Calorie Tracking — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the text-only meal log on the Physical Health page with a per-meal cart of food items (gram-quantified), auto-derive calories/protein/carbs/fats from items, and add calories as a tracked metric.

**Architecture:** Pure-JS helpers in `src/physical-health.js` (food lookup, macro derivation, slugify). All UI stays inline in `App.svelte` (matches existing pattern). Food database is seeded in `data.public.js` and `data.private.js`. Custom foods at runtime persist to localStorage with a "Copy snippet" affordance for promoting into the file. Legacy daily entries (text meals) migrate on load.

**Tech Stack:** Svelte 5 (legacy syntax — no runes), Vite 8, Vitest (added by Task 1) for unit-testing pure helpers, vanilla CSS.

**Spec:** [docs/superpowers/specs/2026-05-07-meal-cart-design.md](../specs/2026-05-07-meal-cart-design.md)

## Svelte 5 / project gotchas

- The codebase uses Svelte 5 in **legacy mode** (no runes). Reactive deps in `$:` blocks track only identifiers referenced directly in the expression — function calls do **not** propagate inner reads. When you compute a derived value, reference reactive identifiers (e.g., `physicalHealthLogs.daily[date]`) directly in the `$:` expression, not via a helper function.
- App state is reassigned (immutable updates), not mutated, so `physicalHealthLogs = { ...physicalHealthLogs, ... }` is the pattern.
- All Physical Health UI lives inline in `App.svelte` per existing pattern — do not extract Svelte components in this plan.

```js
export function itemMacros(item, foodsById) {
  const food = foodsById[item.foodId]
  const grams = Number(item.grams)
  if (!food || !Number.isFinite(grams) || grams <= 0) {
    return { kcal: 0, protein: 0, carbs: 0, fats: 0 }
  }
  const factor = grams / 100
  return {
    kcal: food.kcal * factor,
    protein: food.protein * factor,
    carbs: food.carbs * factor,
    fats: food.fats * factor,
  }
}
```

```js
export function dayMacros(entry, foodsById) {
  const empty = { kcal: 0, protein: 0, carbs: 0, fats: 0 }
  if (!entry || !Array.isArray(entry.meals)) return empty
  return entry.meals.reduce((acc, meal) => {
    if (!Array.isArray(meal.items)) return acc
    return meal.items.reduce((inner, item) => {
      const m = itemMacros(item, foodsById)
      return {
        kcal: inner.kcal + m.kcal,
        protein: inner.protein + m.protein,
        carbs: inner.carbs + m.carbs,
        fats: inner.fats + m.fats,
      }
    }, acc)
  }, empty)
}
```

```js
export function migrateDailyEntry(entry) {
  if (!entry || typeof entry !== 'object') {
    return { water: 0, weight: 0, meals: [] }
  }
  const meals = Array.isArray(entry.meals)
    ? entry.meals.map((meal) => {
        if (Array.isArray(meal.items)) {
          return {
            time: meal.time ?? '00:00',
            label: meal.label ?? '',
            items: meal.items,
          }
        }
        return {
          time: meal.time ?? '00:00',
          label: meal.text ?? meal.label ?? '',
          items: [],
        }
      })
    : []
  return {
    water: Number(entry.water) || 0,
    weight: Number(entry.weight) || 0,
    meals,
  }
}
```

```js
export function findFoodById(foods, id) {
  if (!Array.isArray(foods)) return undefined
  return foods.find((f) => f.id === id)
}

export function mergeFoods(base, overrides) {
  const map = new Map()
  for (const food of base ?? []) map.set(food.id, food)
  for (const food of overrides ?? []) map.set(food.id, food)
  return Array.from(map.values())
}
```

```js
function updateMealField(date, mealIndex, field, value) {
  const existing = getPhysicalHealthEntry(date)
  const next = existing.meals.map((meal, i) =>
    i === mealIndex ? { ...meal, [field]: value } : meal,
  )
  setPhysicalHealthEntry(date, { meals: next })

function updateMealItemGrams(date, mealIndex, itemIndex, rawValue) {
  const grams = rawValue === '' ? 0 : Number(rawValue)

  const existing = getPhysicalHealthEntry(date)
  const meal = existing.meals[mealIndex]
  if (!meal) return
  const nextItems = meal.items.map((item, i) =>

  const nextMeals = existing.meals.map((m, i) =>
```

```js
    ...physicalHealthLogs,
    foods: [...(physicalHealthLogs.foods ?? []), food],
  }
  addFoodToCurrentMeal(unique, 100)
  resetCustomDraft()
}
```

```js
function physicalHealthChartPoints(metric, scale) {
  const range = physicalHealthDateRange(scale)
  const lookupValue = (iso) => {
    const entry = physicalHealthLogs.daily[iso]
    if (!entry) return 0
    if (metric === 'water' || metric === 'weight') {
      return Number(entry[metric]) || 0
    }
    const m = dayMacros(entry, physicalHealthFoodsById)
    if (metric === 'calories') return m.kcal
    return m[metric] || 0
  }

  if (scale === 'year') {
    return range.map((weekStart, idx) => {
      const start = new Date(weekStart)
      let sum = 0
      let count = 0
      for (let i = 0; i < 7; i++) {
        const d = new Date(start)
        d.setDate(d.getDate() + i)
        const iso = d.toISOString().slice(0, 10)
        const value = lookupValue(iso)
        if (value > 0) {
          sum += value
          count++
        }
      }
      return {
        date: weekStart,
        label: `Wk ${idx + 1}`,
        value: count > 0 ? sum / count : 0,
        clickDate: weekStart,
      }
    })
  }

  return range.map((iso) => {
    const value = lookupValue(iso)
    const date = new Date(iso)
    return {
      date: iso,
      label: date.toLocaleDateString('en-IN', { day: 'numeric', month: 'short' }),
      value,
      clickDate: iso,
    }
  })
}
```
